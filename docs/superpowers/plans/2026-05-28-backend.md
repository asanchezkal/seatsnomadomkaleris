# Backend Implementation Plan — Office Seat Reservation

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the Java 21 Spring Boot REST API that replaces Supabase, with Azure AD JWT authentication, PostgreSQL persistence via Flyway, immutable audit logging, and a dev auth bypass for local testing without Azure AD.

**Architecture:** Spring Boot 3.x modular monolith. Controllers are thin — all business logic lives in services. Spring Data JPA manages persistence. Spring Security OAuth2 Resource Server validates Azure AD JWTs in production. When `DEV_AUTH=true`, a `DevAuthFilter` reads `X-Dev-User` header and sets authentication directly, bypassing JWT.

**Tech Stack:** Java 21, Spring Boot 3.3.x, Maven, Spring Security OAuth2 Resource Server, Spring Data JPA, Flyway 10, PostgreSQL 16, H2 (test only), JUnit 5, Mockito, Spring Boot Test, AssertJ

```java
package com.kaleris.seatreservation.service;

import com.kaleris.seatreservation.model.AuditLog;
import com.kaleris.seatreservation.repository.AuditLogRepository;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Propagation;
import org.springframework.transaction.annotation.Transactional;

@Service
public class AuditService {

    private final AuditLogRepository auditLogRepository;

    public AuditService(AuditLogRepository auditLogRepository) {
        this.auditLogRepository = auditLogRepository;
    }

    @Transactional(propagation = Propagation.REQUIRES_NEW)
    public void log(String eventType, String userOid, String resourceId, String detail) {
        auditLogRepository.save(new AuditLog(eventType, userOid, resourceId, detail));
    }
}
```

Note: `Propagation.REQUIRES_NEW` ensures audit events are committed even if the calling transaction rolls back.

```java
package com.kaleris.seatreservation.service;

import com.kaleris.seatreservation.dto.CreateDeskRequest;
import com.kaleris.seatreservation.dto.DeskDto;
import com.kaleris.seatreservation.model.Desk;
import com.kaleris.seatreservation.repository.DeskRepository;
import org.springframework.data.domain.Sort;
import org.springframework.http.HttpStatus;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import org.springframework.web.server.ResponseStatusException;

import java.util.List;

@Service
@Transactional
public class DeskService {

    private final DeskRepository deskRepository;
    private final AuditService   auditService;

    public DeskService(DeskRepository deskRepository, AuditService auditService) {
        this.deskRepository = deskRepository;
        this.auditService   = auditService;
    }

    @Transactional(readOnly = true)
    public List<DeskDto> listDesks() {
        return deskRepository.findAll(Sort.by("id")).stream()
            .map(DeskDto::from)
            .toList();
    }

    public DeskDto addDesk(CreateDeskRequest request, String actorOid) {
        if (deskRepository.existsById(request.id())) {
            throw new ResponseStatusException(HttpStatus.CONFLICT,
                "Desk already exists: " + request.id());
        }
        Desk desk = deskRepository.save(new Desk(request.id(), request.label()));
        auditService.log("DESK_ADDED", actorOid, request.id(), null);
        return DeskDto.from(desk);
    }
```

```java
package com.kaleris.seatreservation.service;

import com.kaleris.seatreservation.dto.CreateReservationRequest;
import com.kaleris.seatreservation.dto.ReservationDto;
import com.kaleris.seatreservation.model.Reservation;
import com.kaleris.seatreservation.repository.DeskRepository;
import com.kaleris.seatreservation.repository.ReservationRepository;
import org.springframework.http.HttpStatus;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import org.springframework.web.server.ResponseStatusException;

import java.time.LocalDate;
import java.util.List;
import java.util.UUID;

@Service
@Transactional
public class ReservationService {

    private final ReservationRepository reservationRepository;
    private final DeskRepository        deskRepository;
    private final AuditService          auditService;

    public ReservationService(ReservationRepository reservationRepository,
                               DeskRepository deskRepository,
                               AuditService auditService) {
        this.reservationRepository = reservationRepository;
        this.deskRepository        = deskRepository;
        this.auditService          = auditService;
    }

    @Transactional(readOnly = true)
    public List<ReservationDto> listReservations(LocalDate date) {
        List<Reservation> found = date != null
            ? reservationRepository.findByDate(date)
            : reservationRepository.findAll();
        return found.stream().map(ReservationDto::from).toList();
    }

    public ReservationDto createReservation(CreateReservationRequest request, String userOid) {
        var desk = deskRepository.findById(request.deskId())
            .orElseThrow(() -> new ResponseStatusException(HttpStatus.NOT_FOUND,
                "Desk not found: " + request.deskId()));

        if (reservationRepository.existsByDeskIdAndDate(request.deskId(), request.date())) {
            throw new ResponseStatusException(HttpStatus.CONFLICT,
                "Desk already reserved for this date");
        }
        if (reservationRepository.existsByUserOidAndDate(userOid, request.date())) {
            throw new ResponseStatusException(HttpStatus.CONFLICT,
                "User already has a reservation for this date");
        }

        Reservation saved = reservationRepository.save(
            new Reservation(desk, request.date(), userOid));
        auditService.log("RESERVATION_CREATED", userOid, saved.getId().toString(), null);
        return ReservationDto.from(saved);
    }

    public void cancelReservation(UUID id, String userOid, boolean isAdmin) {
        Reservation r = reservationRepository.findById(id)
            .orElseThrow(() -> new ResponseStatusException(HttpStatus.NOT_FOUND,
                "Reservation not found: " + id));

        if (!isAdmin && !r.getUserOid().equals(userOid)) {
            throw new ResponseStatusException(HttpStatus.FORBIDDEN,
                "Cannot cancel another user's reservation");
        }

        reservationRepository.delete(r);
        auditService.log("RESERVATION_CANCELLED", userOid, id.toString(), null);
    }

    public void clearAllReservations(String actorOid) {
        reservationRepository.deleteAll();
        auditService.log("RESERVATIONS_CLEARED", actorOid, null, null);
    }
}
```

```java
                .anyRequest().authenticated()

                .jwt(jwt -> jwt.jwtAuthenticationConverter(jwtAuthenticationConverter())));
```

```java
        this.reservationService = reservationService;
    }

    @GetMapping
    public List<ReservationDto> listReservations(
            @RequestParam(required = false) @DateTimeFormat(iso = DateTimeFormat.ISO.DATE) LocalDate date) {
        return reservationService.listReservations(date);
    }

    @PostMapping
    @ResponseStatus(HttpStatus.CREATED)
    public ReservationDto createReservation(@RequestBody @Valid CreateReservationRequest request,
                                             Authentication auth) {
        return reservationService.createReservation(request, extractOid(auth));
    }

    @DeleteMapping("/{id}")
    @ResponseStatus(HttpStatus.NO_CONTENT)
    public void cancelReservation(@PathVariable UUID id, Authentication auth) {
        boolean isAdmin = auth.getAuthorities().stream()
            .anyMatch(a -> a.getAuthority().equals("ROLE_Reservation.AppAdmin"));
        reservationService.cancelReservation(id, extractOid(auth), isAdmin);
    }

    @DeleteMapping
    @ResponseStatus(HttpStatus.NO_CONTENT)
    @PreAuthorize("hasAuthority('ROLE_Reservation.AppAdmin')")
    public void clearAllReservations(Authentication auth) {
        reservationService.clearAllReservations(extractOid(auth));
    }

    private String extractOid(Authentication auth) {
        if (auth.getPrincipal() instanceof Jwt jwt) return jwt.getClaimAsString("oid");
        return auth.getName();
    }
}
```

```java
package com.kaleris.seatreservation.service;

import com.kaleris.seatreservation.repository.ReservationRepository;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.scheduling.annotation.Scheduled;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.time.Instant;
import java.time.temporal.ChronoUnit;

@Service
public class RetentionService {

    private static final Logger log = LoggerFactory.getLogger(RetentionService.class);

    private final ReservationRepository reservationRepository;

    public RetentionService(ReservationRepository reservationRepository) {
        this.reservationRepository = reservationRepository;
    }

    @Scheduled(cron = "0 0 2 * * *") // 02:00 daily
    @Transactional
    public void deleteExpiredReservations() {
        Instant cutoff = Instant.now().minus(365, ChronoUnit.DAYS);
        reservationRepository.deleteByCreatedAtBefore(cutoff);
        log.info("Retention job completed — deleted reservations created before {}", cutoff);
    }
}
```
